import torch
import torch.nn as nn
import triton
import triton.language as tl

@triton.jit
def fused_max_subtract_mean_gelu_kernel(
    x_ptr, out_ptr,
    M, N,
    stride_m, stride_n,
    BLOCK_N: tl.constexpr,
):
    # Each program handles one row
    pid_m = tl.program_id(0)
    
    # Find max along dim=1
    max_val = float('-inf')
    for n_start in range(0, N, BLOCK_N):
        offs_n = n_start + tl.arange(0, BLOCK_N)
        mask = offs_n < N
        x = tl.load(x_ptr + pid_m * stride_m + offs_n * stride_n, mask=mask, other=float('-inf'))
        max_val = tl.maximum(max_val, tl.max(x, axis=0))
    
    # After max, we have single value. Mean of single value = itself
    # So x - mean = 0, and GELU(0) = 0
    result = 0.0
    
    tl.store(out_ptr + pid_m, result)


def gemm_max_sub_gelu(x, weight, bias):
    M, K = x.shape
    N = weight.shape[0]

    # max along dim=1 -> (M, 1)
    # mean along dim=1 of (M, 1) -> (M, 1) with same values

    # GELU(0) -> 0
