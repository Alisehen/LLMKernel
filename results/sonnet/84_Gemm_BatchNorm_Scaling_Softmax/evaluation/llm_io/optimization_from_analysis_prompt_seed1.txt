# Analysis Results

**Bottleneck**: The GEMM kernel incorrectly tiles across N dimension with inner K-loops, destroying data reuse and requiring intermediate materialization. The separate softmax kernel adds unnecessary memory round-trips.

**Optimization Strategy**: Replace with standard tiled GEMM (tile M and N in 2D grid, reduce K) followed by fused row-wise BatchNorm+Scale+Softmax in a single pass, eliminating intermediate storage.

**Implementation Plan**: Implement 2D-tiled GEMM kernel (pid_m, pid_n grid) with proper K-reduction and accumulation. Fuse BatchNorm, scaling, and online softmax (single-pass max/sum reduction) into the GEMM output stage before storing final results. This reduces from 2 kernels with poor memory patterns to 1 optimized kernel.

import torch
import torch.nn as nn
import triton
import triton.language as tl
import math


@triton.jit
def fused_gemm_bn_scale_softmax_kernel(
    a_ptr, b_ptr, bias_ptr, bn_weight_ptr, bn_bias_ptr, bn_mean_ptr, bn_var_ptr,
    scale_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
    bn_eps,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)
    
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    
    # Allocate accumulator for the full row
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # Gemm computation - process in tiles across N dimension
    for n in range(0, N, BLOCK_N):
        curr_offs_n = n + offs_n
        
        # Reset accumulator for this N tile
        tile_acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        
        # K reduction
        a_ptrs = a_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + offs_k[:, None] * stride_bk + curr_offs_n[None, :] * stride_bn
        
        for k in range(0, K, BLOCK_K):
            a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K - k), other=0.0)
            b = tl.load(b_ptrs, mask=(offs_k[:, None] < K - k) & (curr_offs_n[None, :] < N), other=0.0)
            tile_acc += tl.dot(a, b, allow_tf32=True)
            a_ptrs += BLOCK_K * stride_ak
            b_ptrs += BLOCK_K * stride_bk
        
        # Add bias
        bias = tl.load(bias_ptr + curr_offs_n, mask=curr_offs_n < N, other=0.0)
        tile_acc += bias[None, :]
        
        # Batch normalization
        bn_mean = tl.load(bn_mean_ptr + curr_offs_n, mask=curr_offs_n < N, other=0.0)
        bn_var = tl.load(bn_var_ptr + curr_offs_n, mask=curr_offs_n < N, other=0.0)
        bn_weight = tl.load(bn_weight_ptr + curr_offs_n, mask=curr_offs_n < N, other=1.0)
        bn_bias = tl.load(bn_bias_ptr + curr_offs_n, mask=curr_offs_n < N, other=0.0)
        
        tile_acc = (tile_acc - bn_mean[None, :]) / tl.sqrt(bn_var[None, :] + bn_eps)
        tile_acc = tile_acc * bn_weight[None, :] + bn_bias[None, :]
        
        # Scale
        scale = tl.load(scale_ptr)
        tile_acc = scale * tile_acc
        
        # Store in accumulator for softmax
        acc_mask = (offs_m[:, None] < M) & (curr_offs_n[None, :] < N)
        acc = tl.where(acc_mask, tile_acc, acc)
        
        # Store intermediate result for this tile
        c_ptrs = c_ptr + offs_m[:, None] * stride_cm + curr_offs_n[None, :] * stride_cn
        tl.store(c_ptrs, tile_acc, mask=acc_mask)


@triton.jit
def softmax_kernel(
    x_ptr, output_ptr,
    M, N,
    stride_xm, stride_xn, stride_om, stride_on,
    BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)
    
    offs_m = pid_m
    offs_n = tl.arange(0, BLOCK_N)
    
    # Load entire row
    max_val = float('-inf')
    for n in range(0, N, BLOCK_N):
        curr_offs_n = n + offs_n
        x_ptrs = x_ptr + offs_m * stride_xm + curr_offs_n * stride_xn
        x = tl.load(x_ptrs, mask=curr_offs_n < N, other=float('-inf'))
        max_val = tl.maximum(max_val, tl.max(x))
    
    # Compute exp and sum
    sum_exp = 0.0
    for n in range(0, N, BLOCK_N):
        curr_offs_n = n + offs_n
        x_ptrs = x_ptr + offs_m * stride_xm + curr_offs_n * stride_xn
        x = tl.load(x_ptrs, mask=curr_offs_n < N, other=0.0)
        exp_x = tl.exp(x - max_val)
        sum_exp += tl.sum(tl.where(curr_offs_n < N, exp_x, 0.0))
    
    # Normalize and store
    for n in range(0, N, BLOCK_N):
        curr_offs_n = n + offs_n
        x_ptrs = x_ptr + offs_m * stride_xm + curr_offs_n * stride_xn
        x = tl.load(x_ptrs, mask=curr_offs_n < N, other=0.0)
        exp_x = tl.exp(x - max_val)
        output = exp_x / sum_exp
        
        output_ptrs = output_ptr + offs_m * stride_om + curr_offs_n * stride_on
        tl.store(output_ptrs, output, mask=curr_offs_n < N)


def fused_gemm_bn_scale_softmax(x, weight, bias, bn_weight, bn_bias, bn_mean, bn_var, scale, bn_eps):
    M, K = x.shape
    N = weight.shape[0]
    
    # Intermediate output after gemm+bn+scale
    intermediate = torch.empty((M, N), device=x.device, dtype=x.dtype)
    b = weight.t().contiguous()
    
    BLOCK_M = 64
    BLOCK_N = 128
    BLOCK_K = 64
    
    grid = lambda META: (triton.cdiv(M, BLOCK_M),)
    fused_gemm_bn_scale_softmax_kernel[grid](
        x, b, bias, bn_weight, bn_bias, bn_mean, bn_var, scale, intermediate,
        M, N, K,
        x.stride(0), x.stride(1), b.stride(0), b.stride(1), 
        intermediate.stride(0), intermediate.stride(1),
        bn_eps,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    
    # Softmax pass
    output = torch.empty((M, N), device=x.device, dtype=x.dtype)
    grid = lambda META: (M,)
    softmax_kernel[grid](
        intermediate, output,
        M, N,
        intermediate.stride(0), intermediate.stride(1),
        output.stride(0), output.stride(1),
        BLOCK_N=256,
    )
    
    return output
