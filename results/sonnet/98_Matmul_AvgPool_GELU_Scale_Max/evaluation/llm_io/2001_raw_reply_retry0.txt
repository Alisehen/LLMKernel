import torch
import torch.nn as nn
import triton
import triton.language as tl
import math

@triton.jit
def fused_matmul_pool_gelu_scale_max_kernel(
    x_ptr, weight_ptr, bias_ptr, output_ptr,
    M, N, K,
    pool_kernel_size, scale_factor,
    stride_xm, stride_xk,
    stride_wn, stride_wk,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)
    
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_k = tl.arange(0, BLOCK_K)
    
    # Initialize max value for each row
    max_val = tl.full((BLOCK_M,), -float('inf'), dtype=tl.float32)
    
    # Number of pooled outputs
    N_pooled = N // pool_kernel_size
    
    # Iterate over pooled groups
    for n_pool in range(N_pooled):
        # Accumulator for pooled sum
        pool_sum = tl.zeros((BLOCK_M,), dtype=tl.float32)
        
        # Iterate over elements in the pool window
        for pool_idx in range(pool_kernel_size):
            n_idx = n_pool * pool_kernel_size + pool_idx
            offs_n = n_idx
            
            # Compute matmul for this specific output column
            x_ptrs = x_ptr + offs_m[:, None] * stride_xm + offs_k[None, :] * stride_xk
            w_ptrs = weight_ptr + offs_n * stride_wn + offs_k * stride_wk
            
            acc = tl.zeros((BLOCK_M,), dtype=tl.float32)
            
            for k in range(0, K, BLOCK_K):
                mask_k = offs_k < K - k
                x_block = tl.load(x_ptrs, mask=(offs_m[:, None] < M) & (mask_k[None, :]), other=0.0)
                w_block = tl.load(w_ptrs, mask=mask_k, other=0.0)
                
                # Dot product for this column
                acc += tl.sum(x_block * w_block[None, :], axis=1)
                
                x_ptrs += BLOCK_K * stride_xk
                w_ptrs += BLOCK_K * stride_wk
            
            # Add bias
            bias_val = tl.load(bias_ptr + offs_n)
            acc += bias_val
            
            # Accumulate for pooling
            pool_sum += acc
        
        # Average pooling
        pool_avg = pool_sum / tl.cast(pool_kernel_size, tl.float32)
        
        # GELU approximation
        sqrt_2_over_pi = 0.7978845608028654
        x = pool_avg
        tanh_arg = sqrt_2_over_pi * (x + 0.044715 * x * x * x)
        
        exp_2x = tl.exp(2.0 * tanh_arg)
        tanh_val = (exp_2x - 1.0) / (exp_2x + 1.0)
        
        gelu_val = 0.5 * x * (1.0 + tanh_val)
        
        # Scale
        scaled_val = gelu_val * scale_factor
        
        # Update max
        max_val = tl.maximum(max_val, scaled_val)
    
    # Store final max value
    output_ptrs = output_ptr + offs_m
    tl.store(output_ptrs, max_val, mask=offs_m < M)


def fused_forward(x, weight, bias, pool_kernel_size, scale_factor):
    M, K = x.shape
    N = weight.shape[0]
    
    output = torch.empty((M,), device=x.device, dtype=x.dtype)
    
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    
    grid = lambda META: (triton.cdiv(M, META['BLOCK_M']),)
    
    fused_matmul_pool_gelu_scale_max_kernel[grid](
        x, weight, bias, output,
        M, N, K,
        pool_kernel_size, scale_factor,
        x.stride(0), x.stride(1),
        weight.stride(0), weight.stride(1),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    
    return output


class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, pool_kernel_size, scale_factor):
        super(ModelNew, self).__init__()
        self.weight = nn.Parameter(torch.randn(out_features, in_features))
        self.bias = nn.Parameter(torch.randn(out_features))
        self.pool_kernel_size = pool_kernel_size
        self.scale_factor = scale_factor

    def forward(self, x):
        return fused_forward(x, self.weight, self.bias, self.pool_kernel_size, self.scale_factor)
