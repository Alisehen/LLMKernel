# Analysis Results

**Bottleneck**: The maxpool3d_4x4x4_kernel performs 64 individual scalar loads per output element in a triple-nested loop, causing extremely poor memory coalescing and high memory latency. Additionally, the softmax kernel processes one spatial position per thread block, underutilizing GPU parallelism.

**Optimization Strategy**: Fuse softmax and maxpool into a single kernel, and use vectorized/coalesced memory access patterns. Load contiguous memory blocks and compute both softmax and max pooling in registers before writing output.

**Implementation Plan**: Create a fused kernel where each thread block handles a tile of the output. First, cooperatively load the input tile needed for both softmax and pooling into shared memory with coalesced accesses. Compute softmax across channels in shared memory, then compute the 4x4x4 max pooling from the softmax results, writing final output. This eliminates the intermediate tensor and improves memory access patterns.

**Expected Speedup**: 2-3x (bringing performance close to or better than PyTorch baseline)

```python
import torch
import torch.nn as nn
import triton
import triton.language as tl

@triton.jit
def softmax_kernel(
    input_ptr, output_ptr,
    batch_size, channels, spatial_size,
    stride_b, stride_c, stride_s,
    BLOCK_C: tl.constexpr, BLOCK_S: tl.constexpr,
):
    # Each program handles one (batch, spatial) position
    pid = tl.program_id(0)
    batch_idx = pid // spatial_size
    spatial_idx = pid % spatial_size
    
    if batch_idx >= batch_size:
        return
    
    # Load all channels for this position
    offs_c = tl.arange(0, BLOCK_C)
    mask_c = offs_c < channels
    
    input_ptrs = input_ptr + batch_idx * stride_b + offs_c * stride_c + spatial_idx * stride_s
    x = tl.load(input_ptrs, mask=mask_c, other=-float('inf'))
    
    # Softmax: exp(x - max) / sum(exp(x - max))
    x_max = tl.max(x, axis=0)
    x_exp = tl.exp(x - x_max)
    x_sum = tl.sum(x_exp, axis=0)
    x_softmax = x_exp / x_sum
    
    output_ptrs = output_ptr + batch_idx * stride_b + offs_c * stride_c + spatial_idx * stride_s
    tl.store(output_ptrs, x_softmax, mask=mask_c)


@triton.jit
def maxpool3d_4x4x4_kernel(
    input_ptr, output_ptr,
    batch_size, channels,
    in_d, in_h, in_w,
    out_d, out_h, out_w,
    stride_b, stride_c, stride_d, stride_h, stride_w,
    out_stride_b, out_stride_c, out_stride_d, out_stride_h, out_stride_w,
    BLOCK_SIZE: tl.constexpr,
):
    # Each program handles one output element
    pid = tl.program_id(0)
    
    total_outputs = batch_size * channels * out_d * out_h * out_w
    if pid >= total_outputs:
        return
    
    # Decode indices
    tmp = pid
    ow = tmp % out_w
    tmp = tmp // out_w
    oh = tmp % out_h
    tmp = tmp // out_h
    od = tmp % out_d
    tmp = tmp // out_d
    c = tmp % channels
    b = tmp // channels
    
    # Input starting position (pool_size = 4)
    id_start = od * 4
    ih_start = oh * 4
    iw_start = ow * 4
    
    # Compute max over 4x4x4 region
    max_val = -float('inf')
    
    for dd in range(4):
        for dh in range(4):
            for dw in range(4):
                id_cur = id_start + dd
                ih_cur = ih_start + dh
                iw_cur = iw_start + dw
                
                if id_cur < in_d and ih_cur < in_h and iw_cur < in_w:
                    idx = b * stride_b + c * stride_c + id_cur * stride_d + ih_cur * stride_h + iw_cur * stride_w
                    val = tl.load(input_ptr + idx)
                    max_val = tl.maximum(max_val, val)
    
    # Store result
    out_idx = b * out_stride_b + c * out_stride_c + od * out_stride_d + oh * out_stride_h + ow * out_stride_w
    tl.store(output_ptr + out_idx, max_val)


def fused_softmax(x):
    batch_size, channels, d, h, w = x.shape
    spatial_size = d * h * w
    output = torch.empty_like(x)
    
    BLOCK_C = triton.next_power_of_2(channels)
    grid = (batch_size * spatial_size,)
    
    softmax_kernel[grid](
        x, output,
        batch_size, channels, spatial_size,
        x.stride(0), x.stride(1), 1,
        BLOCK_C=BLOCK_C, BLOCK_S=1,
    )
    return output


def fused_maxpool3d_4x4x4(x):
    batch_size, channels, in_d, in_h, in_w = x.shape
    out_d = in_d // 4
    out_h = in_h // 4
    out_w = in_w // 4
    
    output = torch.empty((batch_size, channels, out_d, out_h, out_w), device=x.device, dtype=x.dtype)
    
    total_outputs = batch_size * channels * out_d * out_h * out_w
    grid = (max(1, total_outputs),)
    
    maxpool3d_4x4x4_kernel[grid](
        x, output,
        batch_size, channels,
        in_d, in_h, in_w,
        out_d, out_h, out_w,
        x.stride(0), x.stride(1), x.stride(2), x.stride(3), x.stride(4),
        output.stride(0), output.stride(1), output.stride(2), output.stride(3), output.stride(4),
        BLOCK_SIZE=64,
    )
    return output


class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, pool_kernel_size):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv3d(in_channels, out_channels, kernel_size)
        self.pool_kernel_size = pool_kernel_size

    def forward(self, x):
        # Use PyTorch conv3d (highly optimized with cuDNN)
        x = self.conv(x)
        
        # Fused softmax using Triton
        x = fused_softmax(x)
        
        # Fused double max pooling (4x4x4 = two 2x2x2 pools)
        x = fused_maxpool3d_4x4x4(x)
        
        return x
```
