Looking at the current code, I need to optimize the fused softmax + maxpool kernel. The key issues I see:

1. The softmax is computed per-spatial-position across channels, but the current fused kernel has inefficient memory access patterns
2. The 4x4x4 pooling loop with conditional `if valid` creates divergent execution
3. We can improve by better utilizing shared memory and coalesced access

Let me create an optimized version:

```python
import torch
import torch.nn as nn
import triton
import triton.language as tl


@triton.jit
def fused_softmax_maxpool_kernel_v2(
    input_ptr, output_ptr,
    batch_size, channels,
    in_d, in_h, in_w,
    out_d, out_h, out_w,
    in_stride_b, in_stride_c, in_stride_d, in_stride_h, in_stride_w,
    out_stride_b, out_stride_c, out_stride_d, out_stride_h, out_stride_w,
    BLOCK_C: tl.constexpr,
):
    # Each program handles one (batch, output_d, output_h, output_w) position
    pid = tl.program_id(0)
    
    total_outputs = batch_size * out_d * out_h * out_w
    
    # Decode output spatial indices
    tmp = pid
    ow = tmp % out_w
    tmp = tmp // out_w
    oh = tmp % out_h
    tmp = tmp // out_h
    od = tmp % out_d
    b = tmp // out_d
    
    # Early exit for out of bounds
    out_of_bounds = pid >= total_outputs
    
    # Input starting position (pool_size = 4 for combined 2x2x2 twice)
    id_start = od * 4
    ih_start = oh * 4
    iw_start = ow * 4
    
    # Channel offsets - shared across all operations
    offs_c = tl.arange(0, BLOCK_C)
    mask_c = offs_c < channels
    
    # Initialize max values for all channels
    max_vals = tl.full([BLOCK_C], -float('inf'), dtype=tl.float32)
    
    # Unrolled 4x4x4 pooling region - process all 64 positions
    # Using compile-time constants for better optimization
    for dd in tl.static_range(4):
        id_cur = id_start + dd
        valid_d = id_cur < in_d
        
        for dh in tl.static_range(4):
            ih_cur = ih_start + dh
            valid_dh = valid_d & (ih_cur < in_h)
            
            for dw in tl.static_range(4):
                iw_cur = iw_start + dw
                valid = valid_dh & (iw_cur < in_w) & (~out_of_bounds)
                
                # Compute input offset for this spatial position
                input_offset = b * in_stride_b + id_cur * in_stride_d + ih_cur * in_stride_h + iw_cur * in_stride_w
                input_ptrs = input_ptr + input_offset + offs_c * in_stride_c
                
                # Load all channels (masked load handles invalid positions)
                x = tl.load(input_ptrs, mask=mask_c & valid, other=-float('inf'))
                
                # Compute softmax: exp(x - max) / sum(exp(x - max))
                x_max = tl.max(x, axis=0)
                x_centered = x - x_max
                x_exp = tl.exp(x_centered)
                x_sum = tl.sum(x_exp, axis=0) + 1e-10  # numerical stability
                x_softmax = x_exp / x_sum
                
                # Apply validity mask to softmax result
                x_softmax = tl.where(valid, x_softmax, -float('inf'))
                
                # Update max values
                max_vals = tl.maximum(max_vals, x_softmax)
    
    # Store results for all channels
    output_offset = b * out_stride_b + od * out_stride_d + oh * out_stride_h + ow * out_stride_w
    output_ptrs = output_ptr + output_offset + offs_c * out_stride_c
    tl.store(output_ptrs, max_vals, mask=mask_c & (~out_of_bounds))


@triton.jit
def fused_softmax_maxpool_blocked_kernel(
    input_ptr, output_ptr,
    batch_size, channels,
    in_d, in_h, in_w,
    out_d, out_h, out_w,
    in_stride_b, in_stride_c, in_stride_d, in_stride_h, in_stride_w,
    out_stride_b, out_stride_c, out_stride_d, out_stride_h, out_stride_w,
    BLOCK_C: tl.constexpr,
    BLOCK_SPATIAL: tl.constexpr,
):
    # 2D grid: (spatial_blocks, batch)
    pid_spatial = tl.program_id(0)
    pid_batch = tl.program_id(1)
    
    b = pid_batch
    
    # Decode spatial position
    total_spatial = out_d * out_h * out_w
    spatial_idx = pid_spatial * BLOCK_SPATIAL + tl.arange(0, BLOCK_SPATIAL)
    
    # Process each spatial position in the block
    for s in range(BLOCK_SPATIAL):
        curr_spatial = pid_spatial * BLOCK_SPATIAL + s
        valid_spatial = curr_spatial < total_spatial
        
        tmp = curr_spatial
        ow = tmp % out_w
        tmp = tmp // out_w
        oh = tmp % out_h
        od = tmp // out_h
        
        id_start = od * 4
        ih_start = oh * 4
        iw_start = ow * 4
        
        offs_c = tl.arange(0, BLOCK_C)
        mask_c = offs_c < channels
        
        max_vals = tl.full([BLOCK_C], -float('inf'), dtype=tl.float32)
        
        for dd in tl.static_range(4):
            for dh in tl.static_range(4):
                for dw in tl.static_range(4):
                    id_cur = id_start + dd
                    ih_cur = ih_start + dh
                    iw_cur = iw_start + dw
                    
                    valid = valid_spatial & (id_cur < in_d) & (ih_cur < in_h) & (iw_cur < in_w)
                    
                    input_offset = b * in_stride_b + id_cur * in_stride_d + ih_cur * in_stride_h + iw_cur * in_stride_w
                    input_ptrs = input_ptr + input_offset + offs_c * in_stride_c
                    
                    x = tl.load(input_ptrs, mask=mask_c & valid, other=-float('inf'))
                    
                    x_max = tl.max(x, axis=0)
                    x_exp = tl.exp(x - x_max)
                    x_sum = tl.sum(x_exp, axis=0) + 1e-10
                    x_softmax = x_exp / x_sum
                    
                    x_softmax = tl.where(valid, x_softmax, -float('inf'))
                    max_vals = tl.maximum(max_vals, x_softmax)
        
        output_offset = b * out_stride_b + od * out_stride_d + oh * out_stride_h + ow * out_stride_w
        output_ptrs = output_ptr + output_offset + offs_c * out_stride_c
        tl.store(output_ptrs, max_vals, mask=mask_c & valid_spatial)


def fused_softmax_and_maxpool(x):
    """Fused softmax followed by 4x4x4 max pooling"""
    batch_size, channels, in_d, in_h, in_w = x.shape
    out_d = in_d // 4
    out_h = in_h // 4
    out_w = in_w // 4
    
    output = torch.empty((batch_size, channels, out_d, out_h, out_w), device=x.device, dtype=x.dtype)
    
    BLOCK_C = triton.next_power_of_2(channels)
    BLOCK_C = min(BLOCK_C, 128)  # Cap for register pressure
    
    total_outputs = batch_size * out_d * out_h * out_w
    grid = (max(1, total_outputs),)
    
    fused_softmax_maxpool_kernel_v2[grid](
        x, output,
        batch_size, channels,
        in_d, in_h, in_w,
        out_d, out_h, out_w,
        x.stride(0), x.stride(1), x.stride(2), x.stride(3), x.stride(4),
        output.stride(0), output.stride(1), output.stride(2), output.stride(3), output.stride(4),
        BLOCK_C=BLOCK_C,
    )
    return output


class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, pool_kernel_size):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv3d(in_channels, out_channels, kernel_size)
        self.pool_kernel_size = pool_kernel_size

    def forward(self, x):
        # Use PyTorch conv3d (highly optimized with cuDNN)
        x = self.conv(x)
        
        # Fused softmax and double max pooling (4x4x4 = two 2x2x2 pools)
        x = fused_softmax_and_maxpool(x)
        
        return x
```
