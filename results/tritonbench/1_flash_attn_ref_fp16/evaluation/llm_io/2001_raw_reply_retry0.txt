import torch
import torch.nn as nn
import triton
import triton.language as tl
import math

@triton.autotune(
    configs=[
        triton.Config({'BLOCK_M': 32, 'BLOCK_N': 32}, num_warps=2, num_stages=2),
        triton.Config({'BLOCK_M': 32, 'BLOCK_N': 64}, num_warps=4, num_stages=2),
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 32}, num_warps=4, num_stages=2),
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 64}, num_warps=4, num_stages=2),
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 128}, num_warps=4, num_stages=2),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 64}, num_warps=4, num_stages=2),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 128}, num_warps=8, num_stages=2),
    ],
    key=['seq_len', 'head_dim'],
)
@triton.jit
def flash_attention_kernel(
    Q_ptr, K_ptr, V_ptr, O_ptr,
    stride_qbh, stride_qm, stride_qk,
    stride_kbh, stride_kn, stride_kk,
    stride_vbh, stride_vn, stride_vk,
    stride_obh, stride_om, stride_ok,
    seq_len, head_dim,
    scale,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program IDs
    pid_batch_head = tl.program_id(0)
    pid_m = tl.program_id(1)
    
    # Offsets for the current block of queries
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_k = tl.arange(0, BLOCK_K)
    offs_n = tl.arange(0, BLOCK_N)
    
    # Base pointers for this batch/head
    q_base = Q_ptr + pid_batch_head * stride_qbh
    k_base = K_ptr + pid_batch_head * stride_kbh
    v_base = V_ptr + pid_batch_head * stride_vbh
    o_base = O_ptr + pid_batch_head * stride_obh
    
    # Load Q block [BLOCK_M, BLOCK_K] - coalesced access
    q_ptrs = q_base + offs_m[:, None] * stride_qm + offs_k[None, :] * stride_qk
    q_mask = (offs_m[:, None] < seq_len) & (offs_k[None, :] < head_dim)
    q = tl.load(q_ptrs, mask=q_mask, other=0.0)
    
    # Initialize accumulators
    m_i = tl.zeros([BLOCK_M], dtype=tl.float32) - float('inf')
    l_i = tl.zeros([BLOCK_M], dtype=tl.float32)
    acc = tl.zeros([BLOCK_M, BLOCK_K], dtype=tl.float32)
    
    # Iterate over K, V blocks
    num_blocks_n = tl.cdiv(seq_len, BLOCK_N)
    for block_n_idx in range(0, num_blocks_n):
        start_n = block_n_idx * BLOCK_N
        offs_n_curr = start_n + tl.arange(0, BLOCK_N)
        
        # Load K block [BLOCK_N, BLOCK_K] with coalesced access pattern
        # Then transpose in registers for the dot product
        k_ptrs = k_base + offs_n_curr[:, None] * stride_kn + offs_k[None, :] * stride_kk
        k_mask = (offs_n_curr[:, None] < seq_len) & (offs_k[None, :] < head_dim)
        k_block = tl.load(k_ptrs, mask=k_mask, other=0.0)  # [BLOCK_N, BLOCK_K]
        
        # Transpose K for Q @ K^T computation
        k_t = tl.trans(k_block)  # [BLOCK_K, BLOCK_N]
        
        # Compute attention scores: Q @ K^T -> [BLOCK_M, BLOCK_N]
        qk = tl.dot(q, k_t, allow_tf32=True)
        qk = qk * scale
        
        # Mask out-of-bounds
        qk_mask = (offs_m[:, None] < seq_len) & (offs_n_curr[None, :] < seq_len)
        qk = tl.where(qk_mask, qk, float('-inf'))
        
        # Online softmax update
        m_ij = tl.max(qk, axis=1)
        m_new = tl.maximum(m_i, m_ij)
        
        alpha = tl.exp(m_i - m_new)
        beta = tl.exp(m_ij - m_new)
        
        p_sum = tl.sum(tl.exp(qk - m_ij[:, None]), axis=1)
        l_new = alpha * l_i + beta * p_sum
        
        # Update accumulator
        p = tl.exp(qk - m_new[:, None])
        
        # Load V block [BLOCK_N, BLOCK_K] - coalesced access
        v_ptrs = v_base + offs_n_curr[:, None] * stride_vn + offs_k[None, :] * stride_vk
        v_mask = (offs_n_curr[:, None] < seq_len) & (offs_k[None, :] < head_dim)
        v = tl.load(v_ptrs, mask=v_mask, other=0.0)
        
        # Scale previous accumulator and add new contribution
        acc = acc * alpha[:, None] + tl.dot(p.to(v.dtype), v, allow_tf32=True)
        
        m_i = m_new
        l_i = l_new
    
    # Final normalization
    acc = acc / l_i[:, None]
    
    # Store output
    o_ptrs = o_base + offs_m[:, None] * stride_om + offs_k[None, :] * stride_ok
    o_mask = (offs_m[:, None] < seq_len) & (offs_k[None, :] < head_dim)
    tl.store(o_ptrs, acc.to(tl.float16), mask=o_mask)


def flash_attention(Q, K, V):
    batch, n_heads, seq_len, head_dim = Q.shape
    
    O = torch.empty_like(Q)
    
    scale = 1.0 / math.sqrt(head_dim)
    
    # BLOCK_K must be power of 2 and cover head_dim
    BLOCK_K = triton.next_power_of_2(head_dim)
    
    # Grid: (batch * n_heads, ceil(seq_len / BLOCK_M))
    # BLOCK_M will be determined by autotune
    def grid(meta):
        return (batch * n_heads, triton.cdiv(seq_len, meta['BLOCK_M']))
    
    # Flatten batch and heads for easier indexing
    Q_flat = Q.view(batch * n_heads, seq_len, head_dim)
    K_flat = K.view(batch * n_heads, seq_len, head_dim)
    V_flat = V.view(batch * n_heads, seq_len, head_dim)
    O_flat = O.view(batch * n_heads, seq_len, head_dim)
    
    flash_attention_kernel[grid](
        Q_flat, K_flat, V_flat, O_flat,
        Q_flat.stride(0), Q_flat.stride(1), Q_flat.stride(2),
        K_flat.stride(0), K_flat.stride(1), K_flat.stride(2),
        V_flat.stride(0), V_flat.stride(1), V_flat.stride(2),
        O_flat.stride(0), O_flat.stride(1), O_flat.stride(2),
        seq_len, head_dim,
        scale,
        BLOCK_K=BLOCK_K,
    )
    
    return O


class ModelNew(nn.Module):
    """
    Flash Attention FP16 - Optimized with Triton and autotuning
    """
    def __init__(self):
        super(ModelNew, self).__init__()

    def forward(self, Q: torch.Tensor, K: torch.Tensor, V: torch.Tensor, causal: bool = False) -> torch.Tensor:
        # Ensure inputs are contiguous and on CUDA
        Q = Q.contiguous()
        K = K.contiguous()
        V = V.contiguous()
        
        # Convert to float16 if not already
        if Q.dtype != torch.float16:
            Q = Q.half()
            K = K.half()
            V = V.half()
        
        return flash_attention(Q, K, V)
