# Analysis Results

**Bottleneck**: The kernel converts inputs to FP32 immediately and performs all dot products in FP32 mode. Modern GPUs have tensor cores that can perform FP16 matrix multiplications at 2-4x the throughput of FP32, but this kernel doesn't utilize them effectively.

**Optimization Strategy**: Use FP16 (tl.float16) for the dot product operations while keeping FP32 accumulators for numerical stability. Specifically, keep q, k, v in FP16 for tl.dot operations and cast p to FP16 before the p@v dot product.

**Implementation Plan**: 1) Keep q, k, v loads in FP16 instead of converting to FP32 immediately. 2) Perform qk = tl.dot(q, tl.trans(k)) with FP16 inputs, then scale and convert to FP32 for softmax. 3) Cast p back to FP16 before tl.dot(p, v) to leverage tensor cores. 4) Keep accumulator in FP32 for numerical stability.

**Expected Speedup**: 20-40%

import torch
import torch.nn as nn
import triton
import triton.language as tl
import math

@triton.jit
def flash_attention_kernel(
    Q_ptr, K_ptr, V_ptr, O_ptr,
    stride_qb, stride_qh, stride_qm, stride_qk,
    stride_kb, stride_kh, stride_kn, stride_kk,
    stride_vb, stride_vh, stride_vn, stride_vk,
    stride_ob, stride_oh, stride_om, stride_ok,
    seq_len, head_dim,
    scale,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program IDs
    pid_m = tl.program_id(0)  # which block of queries
    pid_bh = tl.program_id(1)  # batch * n_heads combined
    
    # Compute batch and head indices
    # offs for query block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    
    # Initialize pointers for this batch/head
    Q_block_ptr = Q_ptr + pid_bh * stride_qh + offs_m[:, None] * stride_qm + offs_k[None, :] * stride_qk
    O_block_ptr = O_ptr + pid_bh * stride_oh + offs_m[:, None] * stride_om + offs_k[None, :] * stride_ok
    
    # Initialize accumulators
    m_i = tl.full((BLOCK_M,), float('-inf'), dtype=tl.float32)
    l_i = tl.zeros((BLOCK_M,), dtype=tl.float32)
    acc = tl.zeros((BLOCK_M, BLOCK_K), dtype=tl.float32)
    
    # Load Q block once
    q_mask = (offs_m[:, None] < seq_len) & (offs_k[None, :] < head_dim)
    q = tl.load(Q_block_ptr, mask=q_mask, other=0.0).to(tl.float32)
    
    # Iterate over K, V blocks
    for start_n in range(0, seq_len, BLOCK_N):
        curr_offs_n = start_n + offs_n
        
        # Load K block
        K_block_ptr = K_ptr + pid_bh * stride_kh + curr_offs_n[:, None] * stride_kn + offs_k[None, :] * stride_kk
        k_mask = (curr_offs_n[:, None] < seq_len) & (offs_k[None, :] < head_dim)
        k = tl.load(K_block_ptr, mask=k_mask, other=0.0).to(tl.float32)
        
        # Compute QK^T for this block: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        qk = tl.dot(q, tl.trans(k), allow_tf32=True) * scale
        
        # Mask out-of-bounds
        qk_mask = (offs_m[:, None] < seq_len) & (curr_offs_n[None, :] < seq_len)
        qk = tl.where(qk_mask, qk, float('-inf'))
        
        # Online softmax update
        m_ij = tl.max(qk, axis=1)
        m_new = tl.maximum(m_i, m_ij)
        
        # Compute exp(qk - m_new)
        p = tl.exp(qk - m_new[:, None])
        
        # Update l
        alpha = tl.exp(m_i - m_new)
        l_new = alpha * l_i + tl.sum(p, axis=1)
        
        # Load V block
        V_block_ptr = V_ptr + pid_bh * stride_vh + curr_offs_n[:, None] * stride_vn + offs_k[None, :] * stride_vk
        v_mask = (curr_offs_n[:, None] < seq_len) & (offs_k[None, :] < head_dim)
        v = tl.load(V_block_ptr, mask=v_mask, other=0.0).to(tl.float32)
        
        # Update accumulator: scale old acc and add new contribution
        acc = acc * alpha[:, None] + tl.dot(p.to(tl.float32), v, allow_tf32=True)
        
        # Update m and l
        m_i = m_new
        l_i = l_new
    
    # Final normalization
    acc = acc / l_i[:, None]
    
    # Store output
    o_mask = (offs_m[:, None] < seq_len) & (offs_k[None, :] < head_dim)
    tl.store(O_block_ptr, acc.to(tl.float32), mask=o_mask)


def flash_attention(Q, K, V):
    batch, n_heads, seq_len, head_dim = Q.shape
    
    # Output tensor
    O = torch.empty_like(Q)
    
    # Scale factor
    scale = 1.0 / math.sqrt(head_dim)
    
    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = head_dim  # Must cover full head_dim
    
    # Grid: one program per query block per batch*head
    grid = (triton.cdiv(seq_len, BLOCK_M), batch * n_heads)
    
    # Reshape to merge batch and heads for simpler indexing
    Q_flat = Q.view(batch * n_heads, seq_len, head_dim)
    K_flat = K.view(batch * n_heads, seq_len, head_dim)
    V_flat = V.view(batch * n_heads, seq_len, head_dim)
    O_flat = O.view(batch * n_heads, seq_len, head_dim)
    
    flash_attention_kernel[grid](
        Q_flat, K_flat, V_flat, O_flat,
        Q_flat.stride(0), Q_flat.stride(0), Q_flat.stride(1), Q_flat.stride(2),
        K_flat.stride(0), K_flat.stride(0), K_flat.stride(1), K_flat.stride(2),
        V_flat.stride(0), V_flat.stride(0), V_flat.stride(1), V_flat.stride(2),
        O_flat.stride(0), O_flat.stride(0), O_flat.stride(1), O_flat.stride(2),
        seq_len, head_dim,
        scale,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    
    return O
