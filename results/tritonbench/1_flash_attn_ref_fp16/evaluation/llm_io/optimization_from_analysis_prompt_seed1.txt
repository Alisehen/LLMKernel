@triton.jit
def flash_attention_kernel(
    Q_ptr, K_ptr, V_ptr, O_ptr,
    stride_qbh, stride_qm, stride_qk,
    stride_kbh, stride_kn, stride_kk,
    stride_vbh, stride_vn, stride_vk,
    stride_obh, stride_om, stride_ok,
    seq_len, head_dim,
    scale,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program IDs
    pid_batch_head = tl.program_id(0)
    pid_m = tl.program_id(1)
    
    # Offsets for the current block of queries
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_k = tl.arange(0, BLOCK_K)
    offs_n = tl.arange(0, BLOCK_N)
    
    # Base pointers for this batch/head
    q_base = Q_ptr + pid_batch_head * stride_qbh
    k_base = K_ptr + pid_batch_head * stride_kbh

    o_base = O_ptr + pid_batch_head * stride_obh

    q = tl.load(q_ptrs, mask=q_mask, other=0.0)

    l_i = tl.zeros([BLOCK_M], dtype=tl.float32)

    # Iterate over K, V blocks

        offs_n_curr = start_n + tl.arange(0, BLOCK_N)
        
        # Load K block transposed: [BLOCK_K, BLOCK_N] for Q @ K^T
        k_ptrs = k_base + offs_n_curr[None, :] * stride_kn + offs_k[:, None] * stride_kk

        k = tl.load(k_ptrs, mask=k_mask, other=0.0)

        qk = tl.dot(q, k, allow_tf32=True)
        qk = qk * scale
        
        # Mask out-of-bounds
        qk_mask = (offs_m[:, None] < seq_len) & (offs_n_curr[None, :] < seq_len)
        qk = tl.where(qk_mask, qk, float('-inf'))
        
        # Online softmax update
        m_ij = tl.max(qk, axis=1)
        m_new = tl.maximum(m_i, m_ij)

        beta = tl.exp(m_ij - m_new)
        
        p_sum = tl.sum(tl.exp(qk - m_ij[:, None]), axis=1)
        l_new = alpha * l_i + beta * p_sum

        p = tl.exp(qk - m_new[:, None])
        
        # Load V block [BLOCK_N, BLOCK_K]
        v_ptrs = v_base + offs_n_curr[:, None] * stride_vn + offs_k[None, :] * stride_vk

        v = tl.load(v_ptrs, mask=v_mask, other=0.0)

        l_i = l_new
    
    # Final normalization
    acc = acc / l_i[:, None]
